	\begin{frame}{Basic Principles of Graph Theory}
		\begin{enumerate}
			\item \textbf{Graph:} A graph $G=(V,E)$ consists of a non-empty set $V$ of vertices and a set $E$ of edges, each edge joining a pair of vertices.
			\item \textbf{Degree:} The degree $d(v)$ of a vertex $v$ is the number of edges at $v$.
			\item \textbf{Handshaking lemma:} $\displaystyle\sum_{v\in V}d(v)=2|E|$.
			\item \textbf{Connectedness:} $G$ is connected if every two vertices are joined by a path.
			\item \textbf{Tree:} A tree is a connected graph with no cycles.
		\end{enumerate}
	\end{frame}
