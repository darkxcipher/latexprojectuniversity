	\begin{frame}{Axioms of Quantum Mechanics}
		\begin{enumerate}
			\item \textbf{State:} A system is described by a normalized state vector $|\psi\rangle$ in a Hilbert space.
			\item \textbf{Observables:} Every measurable quantity corresponds to a Hermitian operator $\hat A$.
			\item \textbf{Measurement:} The only possible results are the eigenvalues $a_n$ of $\hat A$, with probability $|\langle a_n|\psi\rangle|^2$.
			\item \textbf{Collapse:} After measuring $a_n$, the state becomes the eigenstate $|a_n\rangle$.
			\item \textbf{Evolution:} $i\hbar\,\dfrac{\partial}{\partial t}|\psi\rangle=\hat H|\psi\rangle$.
		\end{enumerate}
	\end{frame}
